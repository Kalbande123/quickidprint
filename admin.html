// ========================================== // SEVASPOT CENTER - BACKEND (FINAL UPDATED) // ========================================== function parseDate(dateStr) { if(!dateStr) return { isValid: false, daysLeft: 0 }; var cleanStr = String(dateStr).replace(/'/g, "").trim(); var parts = cleanStr.split("/"); if(parts.length !== 3) return { isValid: false, daysLeft: 0 }; var expDate = new Date(parts[2], parts[1] - 1, parts[0]); var today = new Date(); today.setHours(0,0,0,0); var diff = Math.ceil((expDate - today) / (1000 * 60 * 60 * 24)); return { isValid: expDate >= today, daysLeft: diff > 0 ? diff : 0 }; } function formatSafe(val) { if (!val) return "-"; if (val instanceof Date) { return Utilities.formatDate(val, "GMT+5:30", "dd/MM/yyyy"); } return String(val).replace(/'/g, "").trim(); } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var usersSheet = ss.getSheetByName("Users"); var transSheet = ss.getSheetByName("Transactions"); var logSheet = ss.getSheetByName("Service_Log"); if (!transSheet) { transSheet = ss.insertSheet("Transactions"); transSheet.appendRow(["Date", "Name", "Username", "Action", "Remark", "Plan Start", "Expiry Date", "Remaining Days", "Status"]); } if (!logSheet) { logSheet = ss.insertSheet("Service_Log"); logSheet.appendRow(["Date & Time", "Shop Name / User", "Mobile", "Address", "Service Category", "Service Details"]); } var action = e.parameter.action; // 1. LOGIN if (action == 'login') { var loginId = String(e.parameter.username).trim(); var pass = String(e.parameter.password).trim(); var data = usersSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][4]).trim() === loginId && String(data[i][5]).trim() === pass) { var status = String(data[i][10]).trim(); if(status !== "Active") return respond('error', 'तुमचे अकाउंट ब्लॉक (Inactive) करण्यात आले आहे. कृपया ॲडमिनशी संपर्क साधा.'); var expiryDate = data[i][7]; var checkExp = parseDate(expiryDate); if(checkExp.isValid) { usersSheet.getRange(i + 1, 10).setValue(checkExp.daysLeft); return respond('success', 'Login Successful', { name: data[i][0], expireDate: formatSafe(expiryDate), remainingDays: checkExp.daysLeft }); } else { usersSheet.getRange(i + 1, 10).setValue(0); return respond('error', 'तुमचा प्लॅन संपला आहे. कृपया रिन्यू करा.'); } } } return respond('error', 'Invalid Credentials'); } // 2. SERVICE LOG (Service usage details) else if (action == 'submitForm' || action == 'printIdCard') { var loginId = String(e.parameter.username).trim(); var isIDCard = (action == 'printIdCard'); var serviceCategory = isIDCard ? e.parameter.cardType : e.parameter.ghoshnaType; // 🟢 आयडीसाठी प्रिंट ऑप्शन (उदा. 4x6 / A4), घोषणापत्रासाठी फक्त नाव var serviceDetails = isIDCard ? (e.parameter.remark || "Print") : serviceCategory.split(' ')[0] + " घोषणापत्र"; var data = usersSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][4]).trim() === loginId) { var expiryDate = data[i][7]; var checkExp = parseDate(expiryDate); if (checkExp.isValid) { var userName = String(data[i][0]).trim(); var userMobile = String(data[i][3]).trim(); var userAddress = String(data[i][2]).trim(); var timeStr = Utilities.formatDate(new Date(), "GMT+5:30", "dd/MM/yyyy HH:mm:ss"); logSheet.appendRow([timeStr, userName, userMobile, userAddress, serviceCategory, serviceDetails]); return respond('success', 'Service Accessed', { remainingDays: checkExp.daysLeft }); } else { return respond('error', 'तुमच्या प्लॅनची मुदत संपली आहे. कृपया रिन्यू करा.'); } } } return respond('error', 'User not found'); } // 3. ADD USER (0 days) else if (action == 'addUser') { var name = e.parameter.name; var shop = e.parameter.shopName; var address = e.parameter.address; var mobile = String(e.parameter.mobile).trim(); var username = String(e.parameter.username).trim(); var pass = e.parameter.password; var data = usersSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][4]).trim() === username) return respond('error', 'हा युजरनेम आधीपासूनच रजिस्टर आहे.'); if (String(data[i][3]).trim() === mobile) return respond('error', 'हा मोबाईल नंबर आधीपासूनच रजिस्टर आहे.'); } var dateStr = Utilities.formatDate(new Date(), "GMT+5:30", "dd/MM/yyyy"); usersSheet.appendRow([name, shop, address, mobile, username, pass, "'" + dateStr, "'" + dateStr, "'" + dateStr, 0, "Active"]); transSheet.appendRow(["'" + dateStr, name, username, "New Account Created", "No Active Plan", "'" + dateStr, "'" + dateStr, 0, "Active"]); return respond('success', 'नवीन युजर सेव्ह झाला! (लॉगिन करण्यासाठी प्लॅन रिन्यू करा)'); } // 4. RECHARGE & WELCOME PLAN else if (action == 'recharge') { var loginId = String(e.parameter.mobile).trim(); var rechType = String(e.parameter.rechType).trim(); var addDays = parseInt(e.parameter.amount) || 0; var remark = e.parameter.remark || rechType; var data = usersSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][4]).trim() === loginId || String(data[i][3]).trim() === loginId) { var currentExpiry = formatSafe(data[i][7]); var parts = currentExpiry.split("/"); var newExpDate = new Date(); if (rechType === "Zero Days") { newExpDate = new Date(); newExpDate.setDate(newExpDate.getDate() - 1); } else { if(parts.length === 3) { var oldExp = new Date(parts[2], parts[1] - 1, parts[0]); if(oldExp > new Date() && rechType !== "Welcome Plan") newExpDate = oldExp; } if (rechType === "Reduce Days") { newExpDate.setDate(newExpDate.getDate() - addDays); } else { newExpDate.setDate(newExpDate.getDate() + addDays); } } var newExpDateStr = Utilities.formatDate(newExpDate, "GMT+5:30", "dd/MM/yyyy"); var todayStr = Utilities.formatDate(new Date(), "GMT+5:30", "dd/MM/yyyy"); var checkExp = parseDate(newExpDateStr); var planStart = (rechType === "Welcome Plan" || rechType === "Plan Renew") ? todayStr : formatSafe(data[i][6]); usersSheet.getRange(i + 1, 7).setValue("'" + planStart); usersSheet.getRange(i + 1, 8).setValue("'" + newExpDateStr); if(rechType === "Plan Renew" || rechType === "Welcome Plan") { usersSheet.getRange(i + 1, 9).setValue("'" + todayStr); } usersSheet.getRange(i + 1, 10).setValue(checkExp.daysLeft); var userStatus = String(data[i][10]).trim() || "Active"; transSheet.appendRow(["'" + todayStr, data[i][0], data[i][4], rechType, remark, "'" + planStart, "'" + newExpDateStr, checkExp.daysLeft, userStatus]); return respond('success', 'प्लॅन यशस्वीरीत्या अपडेट झाला! उरलेले दिवस: ' + checkExp.daysLeft); } } return respond('error', 'युजर सापडला नाही.'); } // 5. SEARCH USER else if (action == 'searchUser') { var searchId = String(e.parameter.mobile).trim().toLowerCase(); var data = usersSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var sheetMobile = String(data[i][3]).trim().toLowerCase(); var sheetUsername = String(data[i][4]).trim().toLowerCase(); if (sheetMobile === searchId || sheetUsername === searchId) { return respond('success', 'युजर सापडला!', { name: data[i][0], shop: data[i][1], address: data[i][2], mobile: data[i][3], username: data[i][4], pass: data[i][5], startDate: formatSafe(data[i][6]), expireDate: formatSafe(data[i][7]), lastRecharge: formatSafe(data[i][8]), remainingDays: data[i][9], userStatus: data[i][10] }); } } return respond('error', 'युजर सापडला नाही.'); } // 6. UPDATE USER (Block Reason & Days Preservation) else if (action == 'updateUser') { var oldId = String(e.parameter.oldMobile).trim(); var newStatus = String(e.parameter.newStatus).trim(); var blockReason = String(e.parameter.blockReason).trim(); var data = usersSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][4]).trim() === oldId) { var oldStatus = String(data[i][10]).trim(); usersSheet.getRange(i + 1, 1).setValue(e.parameter.newName); usersSheet.getRange(i + 1, 2).setValue(e.parameter.newShop); usersSheet.getRange(i + 1, 3).setValue(e.parameter.newAddress); usersSheet.getRange(i + 1, 4).setValue(e.parameter.newMobile); usersSheet.getRange(i + 1, 5).setValue(e.parameter.newUsername); usersSheet.getRange(i + 1, 6).setValue(e.parameter.newPass); usersSheet.getRange(i + 1, 11).setValue(newStatus); if(oldStatus !== newStatus) { var remarkTxt = (newStatus === "Blocked") ? (blockReason ? "Reason: " + blockReason : "Blocked by Admin") : "Account Unblocked / Active"; var todayStr = Utilities.formatDate(new Date(), "GMT+5:30", "dd/MM/yyyy"); transSheet.appendRow(["'" + todayStr, e.parameter.newName, e.parameter.newUsername, "Status Updated", remarkTxt, "'" + formatSafe(data[i][6]), "'" + formatSafe(data[i][7]), data[i][9], newStatus]); } return respond('success', 'डेटा आणि स्टेटस अपडेट झाले!'); } } return respond('error', 'अपडेट होऊ शकला नाही.'); } // 7. STATEMENT else if (action == 'statement') { var loginId = String(e.parameter.mobile).trim(); var data = transSheet.getDataRange().getValues(); var userTransactions = []; for (var i = 1; i < data.length; i++) { if (String(data[i][2]).trim() === loginId) { userTransactions.push({ date: formatSafe(data[i][0]), actionType: data[i][3], remark: data[i][4], planStart: formatSafe(data[i][5]), expiry: formatSafe(data[i][6]), remaining: data[i][7], status: data[i][8] }); } } return respond('success', 'Statement fetched', { transactions: userTransactions }); } return respond('error', 'Invalid Action'); } function respond(status, message, extraData) { var output = { status: status, message: message }; if (extraData) { for (var key in extraData) { output[key] = extraData[key]; } } return ContentService.createTextOutput(JSON.stringify(output)).setMimeType(ContentService.MimeType.JSON); }
